"""v3 clean-clone rehearsal, one step at a time. Based on Codex's .git/p2-clean.py and
.git/f09-runtime.py; those files are not modified. Usage: v3_clean.py clone|backend|setup|runtime|clients|stop"""

import json
import os
import shutil
import subprocess
import sys
import tempfile
import time
from pathlib import Path
from urllib.request import Request, urlopen
from uuid import uuid4

repo = Path("/Users/comert/Desktop/tbr-quote-assistant")
state = Path(__file__).with_name("v3_clone_path")
tag = "demo-candidate-20260921-v3"
mode = sys.argv[1]


def run(command, cwd, env=None, capture=False):
    print("Command:", " ".join(command), flush=True)
    result = subprocess.run(command, cwd=cwd, env=env, capture_output=capture, text=True)
    if capture:
        print(result.stdout, result.stderr, flush=True)
    print("Exit code:", result.returncode, flush=True)
    if result.returncode:
        raise SystemExit(result.returncode)
    return result.stdout


if mode == "clone":
    dest = Path(tempfile.mkdtemp(prefix="tbr-f09-v3-")) / "repo"
    origin = subprocess.check_output(["git", "remote", "get-url", "origin"], cwd=repo, text=True).strip()
    run(["git", "clone", "--branch", tag, "--single-branch", origin, str(dest)], repo)
    state.write_text(str(dest))
    # New volume names leave the v1/v2 rehearsal data intact.
    (dest / ".git/f09-volume.yaml").write_text(
        "volumes:\n  pgdata:\n    name: tbr-f09-clean-v3-pgdata\n"
        "  test_pgdata:\n    name: tbr-f09-clean-v3-test-pgdata\n"
    )
    sys.exit(0)

dest = Path(state.read_text())
env = dict(
    os.environ,
    COMPOSE_PROJECT_NAME="tbr-f09-clean",
    API_BIND_HOST="127.0.0.1",
    API_PORT="18001",
    WEB_PORT="15173",
    COMPOSE_FILE=str(dest / "compose.yaml") + ":" + str(dest / ".git/f09-volume.yaml"),
)
commit = subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=dest, text=True).strip()
print("Clone:", dest, "Tag:", tag, "Commit:", commit, flush=True)
print("Project tbr-f09-clean; loopback 18001/15173; fresh v3 named volumes; old volumes retained", flush=True)

if mode == "backend":
    # Compose needs the generated .env; README order is init_env, then Compose.
    run(["python3", "scripts/init_env.py"], dest, env)
    run(["docker", "compose", "--profile", "test", "run", "--build", "--rm", "test"], dest, env)
elif mode == "setup":
    run(["python3", "scripts/init_env.py"], dest, env)
    run(["docker", "compose", "up", "--build", "-d", "--wait", "api", "web"], dest, env)
    # --no-deps needs the running db; the first backend-step attempt ran before it existed.
    run(["docker", "compose", "run", "--rm", "--no-deps", "migrate-seed", "alembic", "check"], dest, env)
elif mode == "clients":
    for command in [
        ["npm", "ci"],
        ["npm", "test"],
        ["npm", "run", "typecheck"],
        ["npm", "run", "lint"],
        ["npm", "--prefix", "apps/web", "ci"],
        ["npm", "--prefix", "apps/web", "run", "build"],
        ["npm", "exec", "--workspace", "@tbr/mobile", "--", "expo", "install", "--check"],
        ["python3", "scripts/sync_web_contracts.py", "--check"],
    ]:
        run(command, dest, env)
elif mode == "stop":
    run(["docker", "compose", "--profile", "test", "stop"], dest, env)
elif mode == "runtime":

    def call(port, path, payload=None):
        data = None if payload is None else json.dumps(payload).encode()
        request = Request(
            f"http://127.0.0.1:{port}" + path, data=data, headers={"Content-Type": "application/json"}
        )
        with urlopen(request, timeout=15) as response:
            return json.load(response)

    def compose(*args):
        return run(["docker", "compose", *args], dest, env, capture=True)

    original = call(8001, "/api/quotes/Q-1001")
    counts = compose(
        "exec", "-T", "db", "psql", "-U", "tbr_owner", "-d", "tbr_quotes", "-At", "-c",
        "SELECT (SELECT count(*) FROM products), (SELECT count(*) FROM knowledge_entries), "
        "(SELECT count(*) FROM customers), (SELECT count(*) FROM quotes), "
        "(SELECT count(*) FROM quote_items), (SELECT count(*) FROM price_rules)",
    )
    assert counts.strip() == "48|22|6|10|8|6", counts
    before = call(18001, "/api/quotes/Q-1001")
    assert before["version"] == 1 and before["items"][0]["quantity"] == 1
    session = call(18001, "/api/chat/sessions", {"customer_id": "CUST-IST-001", "quote_id": "Q-1001", "channel": "web"})
    body = {"session_id": session["session_id"], "quote_id": "Q-1001", "message_id": uuid4().hex, "message": "BlueScan Air 1 adet daha ekle."}
    call(18001, "/api/chat", body)
    after = call(18001, "/api/quotes/Q-1001")
    assert after["version"] == 2 and after["items"][0]["quantity"] == 2 and after["net_total_try"] == "15980.00"
    assert call(15173, "/api/quotes/Q-1001") == after
    call(18001, "/api/chat", body)
    assert call(18001, "/api/quotes/Q-1001") == after
    logs = call(18001, "/api/tool-calls?session_id=" + session["session_id"])
    mutations = [x for x in logs if x["tool_name"] == "add_to_quote"]
    assert len(mutations) == 2 and sum(x["replayed"] for x in mutations) == 1 and sum(x["mutation_applied"] for x in mutations) == 1
    policy = call(18001, "/api/chat", dict(body, message_id=uuid4().hex, message="İade süresi nedir?"))
    assert policy["mode"] == "fallback" and policy["provider_calls"] == 0
    assert {"KNE-RET-001", "KNE-FALL-001"} <= {s["source_id"] for s in policy["sources"]}
    assert call(18001, "/api/quotes/Q-1001") == after
    print("PASS original seed counts; real add; direct API and web proxy same quote; replay one effect; sourced keyless policy without mutation", flush=True)
    # Admin key model on the fresh install (updated check copied beside the tagged scripts).
    shutil.copy(repo / "scripts/check_admin_runtime.py", dest / "scripts/check_admin_runtime_current.py")
    run(["python3", "scripts/check_admin_runtime_current.py"], dest, env)
    compose("restart", "db", "api")
    for attempt in range(40):
        try:
            assert call(18001, "/health/ready") == {"status": "ready"}
            break
        except Exception:
            if attempt == 39:
                raise
            time.sleep(1)
    assert call(18001, "/api/quotes/Q-1001") == after
    compose("run", "--rm", "--no-deps", "migrate-seed")
    assert call(18001, "/api/quotes/Q-1001") == after
    call(18001, "/api/chat", body)
    assert call(18001, "/api/quotes/Q-1001") == after
    assert call(15173, "/api/quotes/Q-1001") == after
    assert call(8001, "/api/quotes/Q-1001") == original
    print("PASS API/DB restart persistence; reseed preserves quantity/version; persisted receipt replay; original demo unchanged", flush=True)
